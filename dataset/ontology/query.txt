SELECT ?propertyId ?label ?url ?description
WHERE {
  ?property a wikibase:Property .
  BIND(STRAFTER(STR(?property), "http://www.wikidata.org/entity/") AS ?propertyId)
  
  # Get labels and descriptions in English (change 'en' for different language)
  ?property rdfs:label ?label .
  FILTER(LANG(?label) = "en")
  
  # Get descriptions
  ?property schema:description ?description .
  FILTER(LANG(?description) = "en")
  
  # Create URL
  BIND(STR(?property) AS ?url)
}
ORDER BY ?propertyId
